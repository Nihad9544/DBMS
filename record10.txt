2. Create a procedure to display invoice details of a particular customer.


mysql> DELIMITER //
mysql> CREATE PROCEDURE customer_invoice(in cid int)
    -> BEGIN
    -> SELECT customer.id,
    -> customer.name,
    -> invoice_master.inv_id,
    -> invoice_master.inv_date
    -> FROM customer
    -> JOIN invoice_master
    -> ON customer.id = invoice_master.customer_id
    -> WHERE customer.id = cid;
    -> end
    -> //

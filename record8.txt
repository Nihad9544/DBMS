2. Create a view for invoice products[cname, , inv_id, product_id, quantity]


mysql> CREATE VIEW products AS SELECT customer.name AS cname, invoice_master.inv_id,Invoice_item.inv_id AS product_id,Invoice_item.quantity FROM customer JOIN invoice_master ON customer.id = invoice_master.customer_id JOIN Invoice_item ON invoice_master.inv_id = Invoice_item.inv_id;
